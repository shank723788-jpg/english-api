<script>



let currentUser = null;



// 内置单词库

const words = [

    {en:"apple",zh:"苹果"},

    {en:"banana",zh:"香蕉"},

    {en:"book",zh:"书"},

    {en:"teacher",zh:"老师"},

    {en:"student",zh:"学生"},

    {en:"happy",zh:"开心的"},

    {en:"beautiful",zh:"漂亮的"},

    {en:"friend",zh:"朋友"},

    {en:"learn",zh:"学习"},

    {en:"school",zh:"学校"},

    {en:"water",zh:"水"},

    {en:"animal",zh:"动物"},

    {en:"flower",zh:"花"},

    {en:"computer",zh:"电脑"},

    {en:"music",zh:"音乐"}

];



let wordIndex=0;

let todayWords=[];




// 登录


function login(){


    let name=document.getElementById("username").value.trim();


    let password=document.getElementById("password").value.trim();



    if(!name){

        alert("请输入名字");

        return;

    }



    if(password==="" ){

        password="1234";

    }




    let users=JSON.parse(localStorage.getItem("users")||"[]");



    let user=users.find(

        u=>u.name.toLowerCase()==name.toLowerCase()

    );



    if(!user){


        user={

            name:name,

            password:"1234",

            points:30,

            streak:0,

            lastCheckin:"",

            hours:0,

            pet:"",

            petFood:0,

            badges:[]

        };



        users.push(user);


        localStorage.setItem(

            "users",

            JSON.stringify(users)

        );



    }else{


        if(user.password!==password){

            alert("密码错误");

            return;

        }


    }



    currentUser=user;



    document.getElementById("loginPage").classList.add("hidden");


    document.getElementById("appPage").classList.remove("hidden");



    updateUI();



}






// 更新页面


function updateUI(){



    document.getElementById("hello").innerHTML=

    "你好，"+currentUser.name+" 👋";



    document.getElementById("points").innerHTML=

    currentUser.points;



    document.getElementById("streak").innerHTML=

    currentUser.streak;



    document.getElementById("hours").innerHTML=

    currentUser.hours;



    document.getElementById("mineName").innerHTML=

    currentUser.name;



    document.getElementById("minePoints").innerHTML=

    currentUser.points;



    document.getElementById("badges").innerHTML=

    currentUser.badges.length

    ? currentUser.badges.join(" ")

    :"暂无";



    document.getElementById("date").innerHTML=

    new Date().toLocaleDateString();



    showPet();



}





// 页面切换


function showPage(id){



    let pages=[

        "homePage",

        "petPage",

        "minePage",

        "wordPage"

    ];



    pages.forEach(

        p=>{

            document.getElementById(p)

            .classList.add("hidden");

        }

    );



    document.getElementById(id)

    .classList.remove("hidden");



    if(id==="petPage"){

        showPet();

    }


}






// 打卡



function checkin(){



    let today=

    new Date().toISOString().slice(0,10);



    if(currentUser.lastCheckin===today){


        alert("今天已经打卡啦！");

        return;


    }




    if(currentUser.hours<=0){


        alert("课时不足，请联系老师充值");

        return;


    }




    let yesterday=

    new Date(Date.now()-86400000)

    .toISOString()

    .slice(0,10);




    if(currentUser.lastCheckin===yesterday){


        currentUser.streak++;


    }else{


        currentUser.streak=1;


    }




    currentUser.points+=3;



    if(currentUser.streak%7===0){


        currentUser.points+=10;


        alert("连续7天打卡！奖励10积分🎉");


    }




    currentUser.lastCheckin=today;



    checkBadge();



    saveUser();



    alert("打卡成功 +3积分");



    updateUI();



}






function checkBadge(){



    let s=currentUser.streak;



    if(s>=3 && !currentUser.badges.includes("🥉")){

        currentUser.badges.push("🥉");

    }



    if(s>=7 && !currentUser.badges.includes("🥈")){

        currentUser.badges.push("🥈");

    }



    if(s>=15 && !currentUser.badges.includes("🥇")){

        currentUser.badges.push("🥇");

    }



    if(s>=30 && !currentUser.badges.includes("🏆")){

        currentUser.badges.push("🏆");

    }



}





// 保存


function saveUser(){



    let users=

    JSON.parse(localStorage.getItem("users")||"[]");



    let index=

    users.findIndex(

        u=>u.name===currentUser.name

    );



    users[index]=currentUser;



    localStorage.setItem(

        "users",

        JSON.stringify(users)

    );


}






// 单词


function startWords(){



    todayWords=

    [...words]

    .sort(()=>Math.random()-0.5)

    .slice(0,15);



    wordIndex=0;



    showPage("wordPage");



    showWord();



}




function showWord(){



    if(wordIndex>=todayWords.length){


        alert("完成15个单词！去打卡吧🎉");


        showPage("homePage");


        return;


    }



    let w=todayWords[wordIndex];



    document.getElementById("wordBox")

    .innerHTML=

    w.en+

    "<br><small>"+w.zh+"</small>";


}




function nextWord(){


    wordIndex++;


    showWord();


}







// 宠物



function showPet(){



let box=document.getElementById("petBox");


let info=document.getElementById("petInfo");



if(!currentUser.pet){



box.innerHTML="🥚 🥚 🥚<br>选择你的宠物蛋";



info.innerHTML=

`

<button onclick="choosePet('🐱')">

猫咪

</button>


<button onclick="choosePet('🐶')">

小狗

</button>


<button onclick="choosePet('🐰')">

兔子

</button>

`;



}else{



box.innerHTML=currentUser.pet;



info.innerHTML=

`

<p>饥饿值：${currentUser.petFood}%</p>


<button onclick="feedPet()">

🍎 喂食

</button>

`;


}



}





function choosePet(pet){



currentUser.pet=pet;


currentUser.petFood=50;



saveUser();


updateUI();



}





function feedPet(){



currentUser.petFood+=10;



if(currentUser.petFood>100){


currentUser.petFood=100;


}



currentUser.points-=5;



saveUser();



alert("宠物吃饱啦 ❤️");



updateUI();



}
